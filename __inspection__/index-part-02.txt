async function deriveKey(password,salt){
 const material=await crypto.subtle.importKey("raw",new TextEncoder().encode(password),{name:"PBKDF2"},false,["deriveKey"]);
 return crypto.subtle.deriveKey({name:"PBKDF2",salt,iterations:250000,hash:"SHA-256"},material,{name:"AES-GCM",length:256},false,["encrypt","decrypt"]);
}
function b64(input){
 const arr=input instanceof Uint8Array?input:new Uint8Array(input);
 let out="";
 for(let i=0;i<arr.length;i+=0x8000)out+=String.fromCharCode.apply(null,arr.subarray(i,i+0x8000));
 return btoa(out);
}
function bytes(s){return Uint8Array.from(atob(s),c=>c.charCodeAt(0))}
async function encryptData(password,obj){
 const salt=crypto.getRandomValues(new Uint8Array(16)),iv=crypto.getRandomValues(new Uint8Array(12));
 const key=await deriveKey(password,salt);
 const ct=await crypto.subtle.encrypt({name:"AES-GCM",iv},key,new TextEncoder().encode(JSON.stringify(obj)));
 return {v:1,salt:b64(salt),iv:b64(iv),data:b64(new Uint8Array(ct))};
}
/* Envelope: {v,salt,iv,data}. O salt e o IV viajam junto com o conteudo
   criptografado, portanto qualquer dispositivo que conheca a senha de
   protecao deriva exatamente a mesma chave (PBKDF2-SHA256 250k / AES-GCM).
   Nada especifico do dispositivo entra na derivacao. */
const VAULT_B64=/^[A-Za-z0-9+/]+={0,2}$/;
function parseVault(raw){
 let blob=raw;
 if(typeof blob==="string"){try{blob=JSON.parse(blob)}catch(e){return null}}
 if(!blob||typeof blob!=="object")return null;
 const salt=blob.salt,iv=blob.iv,payload=blob.data;
 if(typeof salt!=="string"||typeof iv!=="string"||typeof payload!=="string")return null;
 if(!salt||!iv||!payload)return null;
 if(!VAULT_B64.test(salt)||!VAULT_B64.test(iv)||!VAULT_B64.test(payload))return null;
 return {v:blob.v||1,salt:salt,iv:iv,data:payload};
}
async function decryptData(password,raw){
 const blob=parseVault(raw);
 if(!blob)throw new Error("VAULT_FORMAT");
 const key=await deriveKey(password,bytes(blob.salt));
 const pt=await crypto.subtle.decrypt({name:"AES-GCM",iv:bytes(blob.iv)},key,bytes(blob.data));
 return JSON.parse(new TextDecoder().decode(pt));
}
/* Somente a falha de autenticacao do AES-GCM significa senha diferente.
   Envelope truncado, texto em vez de objeto ou versao desconhecida sao
   reportados como problema de formato, nunca como senha errada. */
async function decryptVault(password,raw){
 if(!parseVault(raw))return {ok:false,reason:"format"};
 try{
   const value=await decryptData(password,raw);
   return {ok:true,value:Array.isArray(value)?value:[]};
 }catch(e){
   return {ok:false,reason:e&&e.message==="VAULT_FORMAT"?"format":"password"};
 }
}
