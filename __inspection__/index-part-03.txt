 caracteres.";return}

   /* Dispositivo novo com cofre na nuvem: a senha informada precisa abrir o
      conteudo que ja existe na conta. Nunca criamos uma protecao paralela. */
   if(securityMode==="restore"){
     msg.textContent="Baixando dados da conta...";
     let raw=null;
     try{raw=await cloudFetchVault()}
     catch(err){msg.textContent="Não foi possível baixar os dados da conta agora. Verifique a conexão e tente de novo.";return}
     if(!raw){
       setSecurityMode(localStorage.getItem(SECURE_KEY)?"unlock":"setup");
       msg.textContent="Nenhum dado encontrado na nuvem para esta conta.";
       return;
     }
     const res=await decryptVault(p,raw);
     if(!res.ok){
       msg.textContent=res.reason==="format"
         ?"Os dados guardados na nuvem estão em um formato que esta versão não reconhece. Nada foi alterado."
         :"Senha de proteção incorreta para os dados desta conta.";
       password.value="";password.focus();
       return;
     }
     /* Qualquer lancamento antigo nao criptografado deste navegador e
        preservado: entra na uniao em vez de ser descartado. */
     const local=localLegacyData;
     data=(local.length&&typeof mergeRecords==="function")?mergeRecords(res.value,local):res.value;
     window.__financePassword=p;
     unlocked=true;
     try{await secureSave()}catch(err){}
     hideLock();render();resetInactivity();
     if(__syncReady) syncNow(false);
     return;
   }

   if(securityMode==="setup"){
     const p2=password2.value;
     if(p!==p2){msg.textContent="As senhas não coincidem.";return}
     window.__financePassword=p;
     data=localLegacyData;
     try{
       unlocked=true;
       await secureSave();
       hideLock();render();resetInactivity();
       if(__syncReady) syncNow(false);
     }catch(err){
       window.__financePassword=null;unlocked=false;
       msg.textContent="Não foi possível proteger os dados neste navegador.";
     }
     return;
   }

   const raw=localStorage.getItem(SECURE_KEY);
   if(!raw){setSecurityMode("setup");msg.textContent="Crie a proteção deste dispositivo para começar.";return}
   const res=await decryptVault(p,raw);
   if(!res.ok){
     msg.textContent=res.reason==="format"
       ?"Os dados protegidos deste navegador estão inválidos ou incompletos."
       :"Senha incorreta.";
     password.value="";password.focus();
     return;
   }
   data=res.value;
   window.__financePassword=p;unlocked=true;hideLock();render();resetInactivity();
   if(__syncReady) syncNow(false);
 };
 if(stored){showLock("unlock");}
 else{showLock("setup");}
}
