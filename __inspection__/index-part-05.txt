const TOMBSTONE_TTL_MS=180*24*60*60*1000;
function stampRecord(x){if(x&&typeof x==="object")x.updatedAt=new Date().toISOString();return x}
function recordTime(x){
 if(!x)return 0;
 const t=Date.parse((x.deleted?x.deletedAt:x.updatedAt)||"");
 return isNaN(t)?0:t;
}
function mergeRecords(a,b){
 /* Marcas de exclusao antigas sao descartadas antes da uniao para que o
    conjunto nao cresca sem limite. */
 const limit=Date.now()-TOMBSTONE_TTL_MS;
 const vigente=x=>{
   if(!x||!x.id)return false;
   if(!x.deleted)return true;
   const t=Date.parse(x.deletedAt||"");
   return isNaN(t)?true:t>=limit;
 };
 const map=new Map();
 const removed=new Map();
 [...b,...a].filter(vigente).forEach(x=>{
   if(x.deleted){
     removed.set(x.id,{id:x.id,deleted:true,deletedAt:x.deletedAt||new Date().toISOString()});
     return;
   }
   /* Entre duas versoes do mesmo lancamento vence a alteracao mais recente;
      sem data de alteracao (dados antigos) fica a versao deste dispositivo. */
   const cur=map.get(x.id);
   if(!cur||recordTime(x)>=recordTime(cur))map.set(x.id,x);
 });
 /* Uma exclusao vigente sempre vence o registro vivo. */
 removed.forEach((v,k)=>map.set(k,v));
 return Array.from(map.values());
}
