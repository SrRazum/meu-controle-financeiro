async function initCloud(){
 if(__supabase) return __syncReady;
 if(!syncConfigured()){setSyncState("err");return false}
 try{
   __supabase=window.supabase.createClient(window.SUPABASE_URL,window.SUPABASE_PUBLISHABLE_KEY,{
     auth:{autoRefreshToken:true,persistSession:true,detectSessionInUrl:true}
   });
   const {data:{session}}=await __supabase.auth.getSession();
   __syncReady=!!session;
   setSyncState(session?"ok":"");
   __supabase.auth.onAuthStateChange((_event,newSession)=>{
     __syncReady=!!newSession;setSyncState(newSession?"ok":"");
     if(!newSession){__cloudMismatch=false;setRecoverVisible(false)}
     refreshSyncUI();
     if(!newSession)return;
     if(unlocked)syncNow(false);
   });
   refreshSyncUI();
   if(session && unlocked){
     await syncNow(false);
   }
   return __syncReady;
 }catch(e){console.warn("Falha ao iniciar sincronização",e);setSyncState("err");return false}
}
